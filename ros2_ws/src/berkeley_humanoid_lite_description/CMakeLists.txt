cmake_minimum_required(VERSION 3.8)
project(berkeley_humanoid_lite_description)

find_package(ament_cmake REQUIRED)

# A fresh clone has dangling mesh symlinks; say so usefully rather than letting
# install() fail with a bare path.
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/require_mesh_directories.cmake)
require_mesh_directories()

# meshes/ and meshes_dexhand/ are symlinks into source/berkeley_humanoid_lite_assets
# and the dexhand workspace, so the STLs are not duplicated here. The trailing
# slash makes CMake follow the symlink and install the contents.
install(DIRECTORY meshes/ DESTINATION share/${PROJECT_NAME}/meshes)
install(DIRECTORY meshes_dexhand/ DESTINATION share/${PROJECT_NAME}/meshes_dexhand)
# Real files, not a symlink: the left-hand set is generated by mirror_meshes.py
# because dexhandv2_description ships no left variant.
install(DIRECTORY meshes_dexhand_left/
        DESTINATION share/${PROJECT_NAME}/meshes_dexhand_left)
# DexHand V1 forearm and wrist, vendored under ros2_ws/vendor/.
install(DIRECTORY meshes_v1/ DESTINATION share/${PROJECT_NAME}/meshes_v1)
install(DIRECTORY urdf launch rviz config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/read_mount.py DESTINATION lib/${PROJECT_NAME})

ament_package()
